```rust
async fn count_lines_in_file(path: &Path) -> Result<LineStats> {
    let content = fs::read_to_string(path).await?;
    let mut total = 0;
    let mut code = 0;
    let mut comments = 0;
    let mut blank = 0;

    for line in content.lines() {
        total += 1;
        let trimmed = line.trim();
        if trimmed.is_empty() {
            blank += 1;
        } else if trimmed.starts_with("//") {
            comments += 1;
        } else {
            code += 1;
        }
    }

    Ok(LineStats { total, code, comments, blank })
}
```

```rust
enum SessionState {
    Initial,
    Processing { prompt: String },
    Complete { output: String },
    Snapshotting { data: Vec<u8> },
    Error { reason: String },
}

struct Session {
    state: SessionState,
    session_id: String,
}

impl Session {
    async fn transition(&mut self, event: SessionEvent) -> Result<()> {
        self.state = match (self.state.clone(), event) {
            (SessionState::Initial, SessionEvent::UserPrompt(p)) => {
                SessionState::Processing { prompt: p }
            }
            (SessionState::Processing { .. }, SessionEvent::AgentFinished(o)) => {
                SessionState::Complete { output: o }
            }
            _ => return Err("Invalid transition"),
        };
        Ok(())
    }
}
```

```rust
impl Session {
    pub async fn create_snapshot(&self) -> Result<SessionSnapshot> {
        Ok(SessionSnapshot {
            session_id: self.id.clone(),
            created_at: chrono::Utc::now().to_rfc3339(),
            grok_version: env!("CARGO_PKG_VERSION").to_string(),
            current_state: self.state.clone(),
            chat_history: self.history.clone(),
            cwd: self.workspace.get_cwd().await?,
            open_files: self.workspace.get_open_files().await?,
            agent_config: self.agent.config.clone(),
            session_memory: self.memory.serialize().await?,
        })
    }

    pub async fn save_snapshot(&self, path: &Path) -> Result<()> {
        let snapshot = self.create_snapshot().await?;
        let json = serde_json::to_vec(&snapshot)?;
        let mut encoder = GzEncoder::new(File::create(path)?, Compression::default());
        encoder.write_all(&json)?;
        encoder.finish()?;
        Ok(())
    }
}
```

```rust
impl Session {
    pub async fn restore_from_snapshot(path: &Path) -> Result<Session> {
        // 1. Read gzipped file
        let file = File::open(path)?;
        let decoder = GzDecoder::new(file);
        let snapshot: SessionSnapshot = serde_json::from_reader(decoder)?;

        // 2. Reconstruct session
        let mut session = Session::new(snapshot.session_id);
        session.state = snapshot.current_state;
        session.history = snapshot.chat_history;
        session.workspace.set_cwd(&snapshot.cwd).await?;
        session.agent.config = snapshot.agent_config;

        // 3. Restore workspace files
        for file_state in snapshot.open_files {
            session.workspace.restore_file(file_state).await?;
        }

        Ok(session)
    }
}
```

Day 6-7: Data Flow Analysis
---

Key concepts:
- Definition (def): where a variable is assigned
- Use (use): where a variable is read
- Dead code: code that never executes
- Unused variables: defined but never used

Simple analysis algorithm:
1. Find all definitions (let x = ...)
2. Find all uses (print x, x + 1, etc.)
3. Match definitions to uses
4. Report unmatched definitions as unused

```rust
struct Definition {
    name: String,
    line: usize,
    scope: Scope,
}

struct Use {
    name: String,
    line: usize,
    scope: Scope,
}

fn find_unused_variables(code: &str) -> Vec<Definition> {
    let defs = extract_definitions(code);
    let uses = extract_uses(code);
    
    defs.into_iter()
        .filter(|def| !uses.iter().any(|u| u.name == def.name && in_same_scope(&u.scope, &def.scope)))
        .collect()
}
```

Week 3: Refactoring Rules

Day 15-17: Design Refactoring Rule System
---

Architecture:
1. Rule = pattern to match + transformation to apply
2. RuleEngine = applies rules in sequence
3. Safe execution = test before applying

```rust
pub trait RefactoringRule: Send + Sync {
    fn name(&self) -> &str;
    fn description(&self) -> &str;
    fn can_apply(&self, analysis: &AnalysisResult) -> bool;
    fn apply(&self, code: &str, analysis: &AnalysisResult) -> Result<String>;
    fn test(&self, code: &str) -> Result<bool>;  // Verify transformation is safe
}

pub struct RuleEngine {
    rules: Vec<Box<dyn RefactoringRule>>,
}

impl RuleEngine {
    pub async fn apply_rules(&self, code: &str) -> Result<RefactoringPlan> {
        let analysis = analyze_code(code)?;
        
        let mut plan = RefactoringPlan::new();
        for rule in &self.rules {
            if rule.can_apply(&analysis) {
                let transformed = rule.apply(code, &analysis)?;
                if rule.test(&transformed)? {
                    plan.add_step(rule.name().to_string(), transformed);
                }
            }
        }
        Ok(plan)
    }
}
```

```rust
pub struct RemoveDeadCodeRule;

impl RefactoringRule for RemoveDeadCodeRule {
    fn name(&self) -> &str { "remove_dead_code" }
    
    fn can_apply(&self, analysis: &AnalysisResult) -> bool {
        !analysis.unused_functions.is_empty()
    }
    
    fn apply(&self, code: &str, analysis: &AnalysisResult) -> Result<String> {
        let mut result = code.to_string();
        for unused_func in &analysis.unused_functions {
            // Find and remove the function definition
            result = remove_function(&result, &unused_func.name)?;
        }
        Ok(result)
    }
    
    fn test(&self, transformed: &str) -> Result<bool> {
        // Re-parse and verify it's still valid Rust
        let mut parser = Parser::new();
        parser.set_language(tree_sitter_rust::language());
        Ok(parser.parse(transformed, None).is_ok())
    }
}
```

Day 41-42: Safety Testing & Rollback
---

Before applying refactorings:
1. Compile code to verify it still works
2. Run tests if available
3. Create backup
4. If compilation fails, rollback

```rust
pub async fn apply_refactoring_safely(plan: &RefactoringPlan) -> Result<ApplyResult> {
    // 1. Backup original
    let backup = backup_file(&plan.file_path).await?;
    
    // 2. Apply transformations
    let mut content = tokio::fs::read_to_string(&plan.file_path).await?;
    for step in &plan.steps {
        content = step.apply(&content)?;
    }
    
    // 3. Write transformed code
    tokio::fs::write(&plan.file_path, &content).await?;
    
    // 4. Test compilation
    let compile_result = compile(&plan.file_path).await;
    match compile_result {
        Ok(()) => {
            // 5. Run tests
            if let Ok(test_result) = run_tests(&plan.file_path).await {
                if test_result.all_pass {
                    return Ok(ApplyResult::Success);
                }
            }
            // Tests failed, but code compiles - report
            Ok(ApplyResult::CompiledButTestsFailed)
        }
        Err(e) => {
            // Rollback on compile error
            restore_from_backup(&plan.file_path, &backup).await?;
            Err(anyhow::anyhow!("Compilation failed after refactoring: {}", e))
        }
    }
}
```
